param([Parameter(Mandatory=$true)][string]$Mode,[Parameter(Mandatory=$true)][string]$RunName)
$ErrorActionPreference='Stop'
$targetRoot=[IO.Path]::GetFullPath('C:/Users/JasmineJiang/Projects/multiple-ai-agent_SLM_LLM')
$reportRoot=Join-Path $targetRoot 'docs/final_sprint/s3-fix-v1'
$runRoot=[IO.Path]::GetFullPath((Join-Path $reportRoot $RunName))
if (-not $runRoot.StartsWith($reportRoot+'\',[StringComparison]::OrdinalIgnoreCase)) { throw 'Run path escapes report root' }
if (Test-Path -LiteralPath $runRoot) { throw 'Existing diagnostic cannot be overwritten' }
New-Item -ItemType Directory -Path $runRoot -Force | Out-Null
$oldLease=Join-Path $targetRoot 'data/granite_endpoint_leases/2fb24565ac96fa6667ae7aac126b1d1c01bd9ec999315b419e6010a19e5d9461.lease.json'
if (Test-Path -LiteralPath $oldLease) {
    # Explicit reconciliation authorized by the user's audit/repair request; preserve the exact marker.
    $processes=@(Get-CimInstance Win32_Process -Filter 'Name = "ollama.exe" OR Name = "llama-server.exe" OR Name = "ollama_llama_server.exe"')
    if ($processes.Count -ne 0) { throw 'Ollama process remains; cannot reconcile an unknown request' }
    $listeners=@(Get-NetTCPConnection -LocalPort 11434 -State Listen -ErrorAction SilentlyContinue)
    if ($listeners.Count -ne 0) { throw 'Ollama listener remains' }
    $resolved=[IO.Path]::GetFullPath($oldLease)
    $leaseRoot=[IO.Path]::GetFullPath((Join-Path $targetRoot 'data/granite_endpoint_leases'))
    $archived=[IO.Path]::GetFullPath((Join-Path $runRoot 'reconciled_prior_lease.json'))
    if (-not $resolved.StartsWith($leaseRoot+'\',[StringComparison]::OrdinalIgnoreCase) -or -not $archived.StartsWith($runRoot+'\',[StringComparison]::OrdinalIgnoreCase)) { throw 'Invalid reconciliation paths' }
    $hash=(Get-FileHash -LiteralPath $resolved).Hash.ToLower()
    Move-Item -LiteralPath $resolved -Destination $archived
    if ((Get-FileHash -LiteralPath $archived).Hash.ToLower() -ne $hash) { throw 'Archived lease hash mismatch' }
    [pscustomobject]@{recorded_at_utc=[DateTime]::UtcNow.ToString('o');basis='user requested audit and precise repair';
        process_count=0;listener_count=0;source=$resolved;archive=$archived;sha256=$hash} |
        ConvertTo-Json | Set-Content -LiteralPath (Join-Path $runRoot 'reconciliation.json') -Encoding utf8
}
Push-Location -LiteralPath $targetRoot
try {
    & './slm/start_granite.ps1' -OutputDirectory (Join-Path $runRoot 'server')
    & './.venv/Scripts/python.exe' -B (Join-Path $PSScriptRoot 'tmp/diagnose_granite.py') --root $targetRoot --output (Join-Path $runRoot 'probe') --owner (Join-Path $runRoot 'server/server_owner.json') --mode $Mode
    $code=$LASTEXITCODE
} finally { Pop-Location }
exit $code
